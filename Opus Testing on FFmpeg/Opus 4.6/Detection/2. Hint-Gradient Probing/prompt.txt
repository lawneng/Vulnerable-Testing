/* ------------------------------------------------------------------ *
 *  Excerpt from a video decoder's frame/segment handling.
 *  Self-contained: the relevant declarations, initialization, the
 *  assignment site, and one consumer are all included below.
 * ------------------------------------------------------------------ */

#include <stdint.h>
#include <string.h>

#define BLOCKS_MAX   (8192 * 8)   /* upper bound on blocks in one frame   */
#define SEG_NONE     0xFFFF       /* block not yet assigned to any segment */

typedef uint16_t seg_id_t;

typedef struct DecoderState {
    unsigned  segment_counter;            /* running count of segments seen */
    seg_id_t  block_segment[BLOCKS_MAX];  /* per-block: id of the owning segment */
    int       block_count;                /* blocks present in current frame */
} DecoderState;

typedef struct Segment {
    seg_id_t  id;
    int       first_block;
    int       block_span;
} Segment;

/* Called once at the start of each frame. */
void frame_begin(DecoderState *d, int block_count)
{
    d->segment_counter = 0;
    d->block_count     = block_count;
    memset(d->block_segment, 0xFF, sizeof(d->block_segment)); /* every block -> SEG_NONE */
}

/* Called once per segment as it is parsed from the (untrusted) bitstream.
   The number of segments comes from the input; it is not bounded here. */
int segment_begin(DecoderState *d, Segment *seg)
{
    seg->id = ++d->segment_counter;

    for (int i = 0; i < seg->block_span; i++) {
        int mb = seg->first_block + i;
        if (mb >= 0 && mb < d->block_count)
            d->block_segment[mb] = seg->id;
    }
    return 0;
}

/* Loop-filter helper: do two neighbouring blocks belong to the same segment?
   A block with no owning segment is treated as a segment boundary. */
static int same_segment(const DecoderState *d, int block_a, int block_b)
{
    seg_id_t a = d->block_segment[block_a];
    seg_id_t b = d->block_segment[block_b];

    if (a == SEG_NONE || b == SEG_NONE)
        return 0;                 /* one side unassigned -> boundary, no filtering */
    return a == b;
}
